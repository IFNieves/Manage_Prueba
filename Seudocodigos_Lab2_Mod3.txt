3)
START

Declarations
    num1 : num
    num2 : num
    operacion : num
    suma : num
    resta : num
    multi : num
    div : num

Output "Inserte dos numeros: "
Input num1
Input num2

Output "Elija la operacion que desea (1=Suma, 2=Resta, 3=Multiplicación, 4=División): "
Input operacion

suma = num1 + num2
resta = num1 - num2
multi = num1 * num2
div = num1 / num2

If operacion = 1 
    Output "Su resultado es: " + suma
Else If operacion = 2 
    Output "Su resultado es: " + resta
Else If operacion = 3 
    Output "Su resultado es: " + multi
Else If operacion = 4 
    Output "Su resultado es: " + div
Else
    Output "Error, su numero no es aceptable"
End If

STOP

18)

Declarations
    balanceDisponible : num
    retiro : num
    balanceNuevo : num

Output "Inserte el balance total en su cuenta: "
Input balanceDisponible

Output "Su balance disponible es $" + balanceDisponible + ". Inserte la cantidad que desea retirar: "
Input retiro

balanceNuevo = balanceDisponible - retiro

If retiro <= balanceDisponible 
    Output "Retiro de $" + retiro + " permitido."
    Output "Su nuevo balance es: $" + balanceNuevo
Else
    Output "El retiro no es posible, fondos insuficientes."
End If
